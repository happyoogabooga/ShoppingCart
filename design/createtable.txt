CREATE TABLE provider (
    provider_id SERIAL NOT NULL,
    username VARCHAR(32) NOT NULL,
    password VARCHAR(32) NOT NULL,
    name VARCHAR(32) NOT NULL,
    PRIMARY KEY (provider_id)
);

CREATE TABLE users (
    user_id SERIAL NOT NULL,
    username VARCHAR(32) NOT NULL,
    password VARCHAR(32) NOT NULL,
    location VARCHAR(255),
    PRIMARY KEY (user_id)
);

CREATE TABLE album (
    SKU SERIAL NOT NULL,
    title VARCHAR(128) NOT NULL,
    description VARCHAR(255) NOT NULL,
    price DECIMAL(5,2) NOT NULL,
    image VARCHAR(255) NOT NULL,
    PRIMARY KEY (SKU)
);

CREATE TABLE cart(
	cart_id serial not null,
	sku Integer not null,
	user_id Integer not null,
	primary key(cart_id),
	foreign key(sku) references AlbumListings(sku),
	foreign key(user_id) references users(user_id)
);

CREATE TABLE IF NOT EXISTS public.checkout (
    order_number SERIAL NOT NULL,
    total_amount NUMERIC(8,2),
    carts_id INTEGER,
    CONSTRAINT checkout_pkey PRIMARY KEY (order_number),
    CONSTRAINT checkout_carts_id_fkey FOREIGN KEY (carts_id)
        REFERENCES public.carts (carts_id)
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
);

ALTER TABLE public.checkout
    ADD COLUMN IF NOT EXISTS created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ADD COLUMN IF NOT EXISTS shipping_name VARCHAR(128),
    ADD COLUMN IF NOT EXISTS shipping_address VARCHAR(255),
    ADD COLUMN IF NOT EXISTS shipping_address_line2 VARCHAR(255),
    ADD COLUMN IF NOT EXISTS shipping_city VARCHAR(128),
    ADD COLUMN IF NOT EXISTS shipping_region VARCHAR(128),
    ADD COLUMN IF NOT EXISTS shipping_postal_code VARCHAR(32),
    ADD COLUMN IF NOT EXISTS shipping_country VARCHAR(128);

CREATE TABLE IF NOT EXISTS public.checkout_items (
    order_item_id SERIAL PRIMARY KEY,
    order_number INTEGER NOT NULL REFERENCES public.checkout(order_number) ON DELETE CASCADE,
    sku INTEGER NOT NULL,
    title VARCHAR(128) NOT NULL,
    artist VARCHAR(128),
    unit_price DECIMAL(10, 2) NOT NULL,
    quantity INTEGER NOT NULL CHECK (quantity > 0)
);

CREATE TABLE AlbumListings(
	sku serial not null,
	album_id integer not null,
	provider_id integer not null,
	price DECIMAL(10, 2) not null,
	quantity integer not null default 0,
	primary key(sku),
	foreign key(album_id) references album(album_id) on delete cascade,
	foreign key(provider_id) references provider(provider_id) on delete cascade
);
