CREATE TABLE IF NOT EXISTS public.album
(
    album_id serial NOT NULL,
    title character varying(128) NOT NULL,
    description character varying(255) NOT NULL,
    image character varying(255) NOT NULL,
    artist character varying(255),
    CONSTRAINT album_pkey PRIMARY KEY (album_id)
)

CREATE TABLE IF NOT EXISTS public.albumlistings
(
    sku serial NOT NULL,
    album_id integer NOT NULL,
    provider_id integer NOT NULL,
    price numeric(10,2) NOT NULL,
    quantity integer DEFAULT 0,
    CONSTRAINT albumlistings_pkey PRIMARY KEY (sku),
    CONSTRAINT albumlistings_album_id_fkey FOREIGN KEY (album_id)
        REFERENCES public.album (album_id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE,
    CONSTRAINT albumlistings_provider_id_fkey FOREIGN KEY (provider_id)
        REFERENCES public.provider (provider_id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE,
    CONSTRAINT check_quantity_non_negative CHECK (quantity >= 0)
)

CREATE TABLE IF NOT EXISTS public.cartitems
(
    cartitem_id serial NOT NULL,
    carts_id integer NOT NULL,
    sku integer NOT NULL,
    CONSTRAINT cart_items_pkey PRIMARY KEY (cartitem_id),
    CONSTRAINT cart_items_carts_id_fkey FOREIGN KEY (carts_id)
        REFERENCES public.carts (carts_id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
    CONSTRAINT cart_items_sku_fkey FOREIGN KEY (sku)
        REFERENCES public.albumlistings (sku) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
)

CREATE TABLE IF NOT EXISTS public.carts
(
    carts_id serial NOT NULL,
    user_id integer NOT NULL,
    CONSTRAINT carts_pkey PRIMARY KEY (carts_id),
    CONSTRAINT carts_user_id_key UNIQUE (user_id),
    CONSTRAINT carts_user_id_fkey FOREIGN KEY (user_id)
        REFERENCES public.users (user_id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
)

CREATE TABLE IF NOT EXISTS public.checkout
(
    checkout_id serial NOT NULL,
    user_id integer NOT NULL,
    shipping_address character varying(256) COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT checkout_pkey PRIMARY KEY (checkout_id),
    CONSTRAINT checkout_user_id_fkey FOREIGN KEY (user_id)
        REFERENCES public.users (user_id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
)

CREATE TABLE IF NOT EXISTS public.checkoutitems
(
    checkoutitems_id serial NOT NULL,
    checkout_id integer NOT NULL,
    sku integer,
    title character varying(128) NOT NULL,
    quantity integer NOT NULL,
    unit_price numeric(10,2) NOT NULL,
    CONSTRAINT checkoutitems_pkey PRIMARY KEY (checkoutitems_id),
    CONSTRAINT checkoutitems_checkout_id_fkey FOREIGN KEY (checkout_id)
        REFERENCES public.checkout (checkout_id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE,
    CONSTRAINT checkoutitems_sku_fkey FOREIGN KEY (sku)
        REFERENCES public.albumlistings (sku) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE SET NULL,
    CONSTRAINT checkoutitems_quantity_check CHECK (quantity > 0)
)

CREATE TABLE IF NOT EXISTS public.provider
(
    provider_id serial NOT NULL,
    username character varying(32) NOT NULL,
    password character varying(32) NOT NULL,
    name character varying(32) NOT NULL,
    CONSTRAINT provider_pkey PRIMARY KEY (provider_id)
)

CREATE TABLE IF NOT EXISTS public.users
(
    user_id serial NOT NULL,
    username character varying(32) NOT NULL,
    password character varying(32)  NOT NULL,
    location character varying(255),
    name character varying(255),
    CONSTRAINT users_pkey PRIMARY KEY (user_id)
)
